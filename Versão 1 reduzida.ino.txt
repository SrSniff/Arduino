int leituraA0 = 0;
int leituraA1 = 0;
float R2 = 0.0; 

void setup() 
{
  Serial.begin(9600); // Inicia comunicação serial (para monitor do IDE do Arduino)
}

void loop() 
{
  // Efetua as leituras analógicas (ADC retorna valor entre 0 e 1023)
  leituraA0 = analogRead(A0);
  leituraA1 = analogRead(A1);

  R2 = 1000 *( (float)leituraA1 / (float)(leituraA0 - leituraA1));
  Serial.print("\tResistência: "); Serial.println(R2);

  delay(600); // 600ms de atualização 
}
